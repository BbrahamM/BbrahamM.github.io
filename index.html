Synchronous scene and asset loading caused main-thread hitches, and manual dialogue/audio binding across thousands of lines created production bottlenecks.

// Non-blocking additive background scene loading with event notification to prevent main-thread stalls on Switch
private IEnumerator LoadSceneBackground(string sceneToLoad, int sceneIndex = 0)
{
    int currentNumberOfScenes = SceneManager.sceneCount;
    AsyncOperation asyncLoad = SceneManager.LoadSceneAsync(sceneToLoad, LoadSceneMode.Additive);

    while (!asyncLoad.isDone || SceneManager.sceneCount == currentNumberOfScenes)
    {
        yield return null;
    }

    backgroundsLoaded[sceneIndex] = SceneManager.GetSceneByName(sceneToLoad);
    OnSceneLoaded(EventArgs.Empty);
}

protected virtual void OnSceneLoaded(EventArgs e)
{
    EventHandler handler = loadSceneHandler;
    handler?.Invoke(this, e);
}

2. Modular Combat Architecture & Pluggable Skill Systems

妖百景 (One Hundred Ghost Scenes)

50+ Items/Skills
Added with Zero C# Recompilation

Combat Math Isolated from VFX/UI

The Challenge: Engineering a scalable turn-based RPG combat and inventory architecture where 50+ items, equipment pieces, and skill formulas could be integrated and iterated upon without creating tight coupling between combat math, Cinemachine cameras, UI presenters, and particle VFX.

The Solution: Isolated turn sequencing (BtlTurnBaseManager), pluggable skill definitions (BtlSkill composing BtlASkillFormula via the Strategy Pattern and UnityEvent callbacks), and atomic item reservations (BtlItemPool). Automated external spreadsheet ingestion into runtime Unity JSON schemas using Python.

BtlSkill.cs — ScriptableObject Skill Definition, Strategy Formula Injection & AI Scoring C# / 妖百景 Production Code 
using System;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Events;

[CreateAssetMenu(fileName = "BtlSkill", menuName = "Scriptable Objects/Battle Manager/Skills/Skill")]
public class BtlSkill : ScriptableObject
{
    [SerializeField] private SKName skillName;
    [SerializeField] private SkillKind skillKind;
    [SerializeField] private CharacterKind whatAttacks;
    [SerializeField] private SkillEffect skillEffect;
    [SerializeField] private List<Use> whereToUse;
    [SerializeField] private float skillAttackPower = 0f;
    [SerializeField] private float skillManaCost = 0f;
    [SerializeField] private float skillOrderOfAction = 3f;
    [SerializeField] private UnityEvent<BtlMCharacterManager, BtlMCharacterManager> applyDamage;

    [HideInInspector] public BtlMCharacterManager characterManager;

    [SerializeField] private string[] modifiers;
    [SerializeField] private SkillDescription[] description;
    [SerializeField] private GameObject effect;
    [SerializeField] private int targetNumber = 1;
    [SerializeField] private bool targetAll = false;
    [SerializeField] private bool alive = true;
    [SerializeField] private bool stackable = false;
    [SerializeField] private SkillExtraEffect statusEffectConfig = new();
    [SerializeField] private BtlASkillFormula formula;

    public string SKNAME => skillName.NAMES[(int)BattleManager._instance.BMLANGUAGE];
    public string SKENNAME => skillName.NAMES[0];
    public SkillKind SkillKind { get => skillKind; set => skillKind = value; }
    public float SKPWR => skillAttackPower;
    public float SKMPCOST => skillManaCost;

    public void Apply(BtlMCharacterManager target)
    {
        applyDamage?.Invoke(BattleManager._instance.Attacker, target);
    }

    public float Test(BtlMCharacterManager target)
    {
        return formula.Test(BattleManager._instance.Attacker, target);
    }

    public string[] SKModifiers { get => modifiers; set => modifiers = value; }
    public CharacterKind SKMWhatAttacks => whatAttacks;
    public SkillDescription[] SKDescriptions => description;
    public GameObject SKANIM => effect;
    public int SKTargetNumber => targetNumber;
    public bool SKTargetAll => targetAll;
    public bool SKAlive => alive;
    public SkillEffect SKEFFECT => skillEffect;
    public SkillExtraEffect SKEXTRAEFFECT => statusEffectConfig;
    public bool SKSTACKABLE => stackable;
    public float SKORDEROFACTION => skillOrderOfAction;
    public List<Use> SKWHERETOUSE => whereToUse;
    public BtlASkillFormula FORMULA => formula;
}

[Serializable]
public class SkillDescription
{
    public DescriptionType WhereToUse = DescriptionType.MAP;
    public string Description = string.Empty;
}

public enum DescriptionType { MAP, BATTLE }
public enum SkillKind { BASIC, MAGIC, SPECIAL, ITEM, ESCAPE }
public enum SkillEffect { ATTACK, HEAL, BUFF, DEBUFF, CROWDCONTROL, DEFENSE }

[Serializable]
public class SkillExtraEffect
{
    public string effectKey = "normal";
    public float baseScoreValue = 1f;
    public bool stackable = false;
    public float duplicatePenalty = 0.5f;
    public float duration = 1f;
    public float potency = 1f;
    public string thinking = "none";
}

[Serializable]
public class SKName
{
    public string[] NAMES;
}

3. Rendering Optimization & Custom Mesh-Combining Tooling

Fate of Windshift & Windshift

Shipped on PC (Steam) | C#, Unity EditorWindow, Mesh API, Draw Call Batching, Frame Debugger Profiling
-75% Batches
Draw Calls (~4,000 to <1,000)
Locked 60 FPS
Open-World 3D Scene Baseline

The Challenge: Dense 3D open-world environments suffered from severe CPU render-thread bottlenecks caused by thousands of unbatched static props and fragmented multi-material submeshes, pushing draw calls above 4,000 per frame.

The Solution: Authored a custom C# Unity Editor utility using the Mesh.CombineMeshes API to spatially group static geometry, merge shared material submeshes, and preserve lightmap UVs and colliders non-destructively—slashing draw calls under 1,000 batches and locking a steady 60 FPS.
